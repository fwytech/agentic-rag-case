# 问题1: ChromaDB安装失败
# 解决: 安装C++编译工具
#   Windows: 安装 Visual C++ Build Tools
#   Linux: sudo apt-get install build-essential
#   macOS: xcode-select --install

# 问题2: DuckDuckGo搜索超时
# 解决: 设置代理或使用VPN
#   export HTTPS_PROXY=http://proxy.example.com:8080

# 解决: 升级到最新版本
#   pip install --upgrade openai
